def softmax(Z):
    exp_Z = np.exp(Z - np.max(Z, axis=1, keepdims=True)) # softmax should be computed on each training example
    return exp_Z / np.sum(exp_Z, axis=1, keepdims=True)

def linear_function(X, theta):
    return np.dot(X, theta)

def categorical_cross_entropy(ytrue, ypred):
    return -np.mean(np.sum(ytrue * np.log(ypred + 1e-12), axis=1))

def initialize_theta(F):
    # 10 classes 0-9
    return np.zeros([F, 10])

def update_function(theta, grads, learning_rate):
    theta = theta - learning_rate * grads
    return theta

def batch_gradient(X, y, ypred):
    'Computes del J(theta) for the current theta'
    N = X.shape[0]
    return (1.0 / N) * np.dot(X.T, (ypred - y))

def train_batch_gd(X, y, num_epochs, learning_rate = 0.1):
    '''
    - Predict
    - Compute the loss
    - Call batch_gradient to get the gradient
    - Update theta
    - Record the loss
    '''
    
    N, F = X.shape  # N number of training examples, F number of features
    losses = [] # stores J(theta) for each epoch
    theta = initialize_theta(F) 

    for epoch in range(num_epochs):
        ypred = softmax(linear_function(X, theta))
        loss = categorical_cross_entropy(y, ypred) # J(theta)
        grads = batch_gradient(X, y, ypred)  # del J(theta)
        theta = update_function(theta, grads, learning_rate)

        losses.append(loss)
        print(f"\nEpoch {epoch}, loss {loss}")
    return losses, theta

ADAM¶

  • Adaptive Moment Estimation
  • Combines
    • Momentum and Adaptive scaling (average squared gradient)
  • $m_t$: memory of direction
  • $v_t$: memory of squared gradients

def get_memory_direction(memory, beta, grads):
    memory = beta * memory + (1 - beta) * grads
    return memory

def get_squared_grad(vt, beta, grads):
    vt = vt * beta + (1-beta) * grads * grads
    return vt

def train_adam(X, y, num_epochs, learning_rate = 0.001, batch_size = 64):
    N, F = X.shape  # N number of training examples, F number of features
    losses = [] # stores J(theta) for each epoch
    theta = initialize_theta(F) 
    epoch = 0
    loss_tolerance = 0.001
    avg_loss = float("inf")
    beta1 = 0.9
    beta2 = 0.999
    mt = 0
    vt = 0
    t = 0

    loader = make_loader(X, y, batch_size)
    while epoch < num_epochs and avg_loss > loss_tolerance:
        running_loss = 0.0
        for X_sample, y_sample in loader:
            X_sample, y_sample = X_sample.numpy(), y_sample.numpy()

            ypred = softmax(linear_function(X_sample, theta))
            loss = categorical_cross_entropy(y_sample, ypred) # J(theta)
            running_loss += loss
            grads = batch_gradient(X_sample, y_sample, ypred)  # del J(theta)

            mt = get_memory_direction(mt, beta1, grads)
            vt = get_squared_grad(vt, beta2, grads)

            t += 1
            mt_hat = mt / (1 - beta1 ** t)
            vt_hat = vt / (1 - beta2 ** t)
            update_term = mt_hat / (np.sqrt(vt_hat) + 1e-8)
            theta = update_function(theta, update_term, learning_rate)

        avg_loss = running_loss / len(loader)
        losses.append(avg_loss)

        print(f"\nEpoch {epoch}, loss {avg_loss}")
        epoch += 1
    return losses, theta

def evaluate_model(model, X, y):
    model.eval() # turn dropout off
    with torch.no_grad(): # do not track gradients
        out = model(X)

    # pick highest scoring digit for image, compare to true label, turn true/false into 1/0 then average, convert result from tensor to number
    accuracy = (out.argmax(1) == y).float().mean().item() 
    return ce(out, y).item(), accuracy

    g = torch.Generator().manual_seed(seed)

def train_no_reg(model, epochs = 30, learning_rate = 0.001):
    train_loader = make_train_loader()
    
    optimizer = torch.optim.Adam(
        model.parameters(),
        lr = learning_rate
    )

    history = {
        "training_loss": [],
        "validation_loss": [],
        "training_accuracy": [],
        "validation_accuracy": []
    }

    for epoch in range(epochs):
        model.train()

        for X_batch, y_batch in train_loader:
            optimizer.zero_grad()

            predictions = model(X_batch)
            loss = ce(predictions, y_batch)

            loss.backward()
            optimizer.step()

def train_l2(model, epochs = 30, learning_rate = 0.001, lam=1e-3):

        for X_batch, y_batch in train_loader:
